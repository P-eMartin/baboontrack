 45def get_last_tracks(all_tracks):
 46    '''
 47    Get the last tracks from the tracking buffer.
 48
 49    Args:
 50        all_tracks: list of list of dict, the tracking buffer containing the tracks for each frame
 51
 52    Returns:
 53        list of dict, the last tracks from the tracking buffer
 54    '''
 55    unique_tracks = []
 56    track_ids = []
 57    for detections in reversed(all_tracks):
 58        for detection in detections:
 59            if detection['track_id'] not in track_ids:

 64def compute_iou(boxA, boxB, image_size):
 65    '''
 66    Compute the Intersection over Union (IoU) of two bounding boxes.
 67
 68    Args:
 69        boxA: list of int, the first bounding box in the format [x, y, w, h]
 70        boxB: list of int, the second bounding box in the format [x, y, w, h]
 71        image_size: list of int, the size of the image in the format [width, height]
 72
 73    Returns:
 74        float, the IoU value
 75    '''
 76    xA = max(boxA[0], boxB[0])*image_size[0]
 77    yA = max(boxA[1], boxB[1])*image_size[1]
 78    xB = min(boxA[0] + boxA[2], boxB[0] + boxB[2])*image_size[0]
 79    yB = min(boxA[1] + boxA[3], boxB[1] + boxB[3])*image_size[1]
 80    interArea = max(0, xB - xA) * max(0, yB - yA)
 81    boxAArea = boxA[2] * boxA[3]* image_size[0] * image_size[1]
 82    boxBArea = boxB[2] * boxB[3]* image_size[0] * image_size[1]
 83    iou = interArea / float(boxAArea + boxBArea - interArea)
 84    return iou

 86def process_detections(detections, last_tracks, track_id, image_size, score, iou_threshold=0.3, log=None):
 87    '''
 88    Process each detection and assign a track ID based on the last tracks and IoU threshold.
 89    Visibility is set to 1 if no overlap with other tracks, and decreases as the IoU with other tracks increases.
 90
 91    Args:
 92        detections: list of dict, the detections for the current frame
 93        last_tracks: list of dict, the last tracks from the tracking buffer
 94        track_id: int, the current track ID
 95        image_size: list of int, the size of the image in the format [width, height]
 96        score: float, the minimum detection score
 97        iou_threshold: float, the IoU threshold for matching detections with tracks
 98        log: logger, the logger to print the information (default None)
 99    Returns:
100        int, the updated track ID after assignment
101    '''
102    to_delete = []
103    matches = []
104    for idx, detection in enumerate(detections):
105        # Remove detections with low score
106        if detection.get('score', detection.get('conf', 0)) < score:
107            print_and_log('Detection with low score found (frame %d): %f' % (idx, detection['conf']), log=log)
108            to_delete.append(idx)
109            continue
110        # Match track ID based on IoU with last tracks
111        best_iou = iou_threshold
112        for track in last_tracks:
113            if 'segmentation' in track and 'segmentation' in detection:
114                iou = compute_mask_iou(track, detection)
115            else:
116                iou = compute_iou(detection['bbox'], track['bbox'], image_size)
117            if iou > best_iou:
118                matches.append({'det_idx':idx, 'track_id': track['track_id'], 'iou': iou})
119                best_iou = iou
120        # Remove existing track_id if exists to avoid confusion with the new assigned track_id
121        if 'track_id' in detection:
122            del detection['track_id']
123
124    # Sort by best IoU first
125    matches = sorted(matches, key=lambda x: x['iou'], reverse=True)
126
127    assigned_dets = set()
128    assigned_tracks = set()
129
130    # Assign globally best matches
131    for match in matches:
132        det_idx = match['det_idx']
133        track_id_match = match['track_id']
134
135        if det_idx in assigned_dets or track_id_match in assigned_tracks:
136            continue
137
138        detections[det_idx]['track_id'] = track_id_match
139        # Visibility
140        detections[det_idx]['visibility'] = get_value_with_precision(1 - match['iou'])
141
142        assigned_dets.add(det_idx)
143        assigned_tracks.add(track_id_match)
144
145    # Assign new track IDs to unmatched detections
146    for idx, detection in enumerate(detections):
147
148        if idx in to_delete:
149            continue
150
151        if 'track_id' not in detection:
152            detection['track_id'] = track_id
153            track_id += 1
154
155        # Megadetector + coco format compatibility
156        detection['category_id'] = int(detection.get('category', detection.get('category_id', 1)))
157        detection['score'] = detection.get('score', detection.get('conf', 0))
158        detection['visibility'] = get_value_with_precision(
159            1-max([match['iou'] for match in matches if match['det_idx'] == idx and match['track_id'] != detection['track_id']], default=0)
160        )
161        # Remove the normalized keys
162        if 'category' in detection: del detection['category']
163        if 'conf' in detection: del detection['conf']
164
165    # Remove detections with low score
166    for idx in reversed(to_delete):
167        del detections[idx]
168
169    return track_id

258def track(my_video, detection_dict, output_file, device='cpu', tracking_size=60, score_th=0.5, tracker_type=None, image_size=None, log=None):
259    '''
260    Track the Baboons in the video using the detection results.
261
262    Args:
263        my_video: VideoFrameIterator, the video to process
264        detection_dict: dict, the detection results
265        output_file: str, the path to save the output file
266        device: str, the device to use for the tracking
267        tracking_size: int, the size of the tracking buffer in number of frames (default 30)
268        score_th: float, score threshold for detections to be considered for tracking (default 0.5)
269        tracker_type: str, the type of tracker to use (default None)
270        image_size: tuple, the size of the image (width, height)
271        log: logger, the logger to print the information (default None)
272
273    Returns:
274        dict, the tracking results
275    '''
276    # Initialization
277    ## Check if the output file already exists
278    if os.path.exists(output_file):
279        print_and_log('Output file %s already exists. Skipping tracking. Loading existing file.' % (output_file), log=log)
280        return output_file
281    
282    ## Check if detection_dict is filepath or dict
283    detection_dict = load_as_detection_dict(detection_dict, image_size=image_size, log=log)
284
285    start_time = time.time()
286    if tracker_type == "bytetrack":
287        from .bt_utils.bytetrack.byte_tracker import BYTETracker
288        print_and_log('Tracking with ByteTrack', log=log)
289        feat_model = None
290        tracker = BYTETracker()
291    elif tracker_type == "deepsort":
292        print_and_log('Tracking with DeepSort', log=log)
293        ## Feature extractor
294        feat_model = ReIDModel(device=device)
295        ## Tracker
296        tracker = init_tracker(max_cosine_distance=0.5, nn_budget=100, max_iou_distance=0.5, max_age=tracking_size, n_init=0)
297    elif tracker_type == "botsort":
298        print_and_log('Tracking with BoTSORT', log=log)
299        from .bt_utils.bot_sort.bot_sort import BoTSORT
300        feat_model = ReIDModel(device=device)
301        tracker = BoTSORT(encoder=feat_model)
302    elif tracker_type == "IoU":
303        print_and_log('Tracking with IoU', log=log)
304        track_id = 0
305        tracking_buffer = []
306        last_tracks = []
307    else:
308        # None - return the detection results without tracking
309        print_and_log('No tracking, just returning detection results', log=log)
310        return detection_dict
311
312    # Loop over the video frames and detection results
313    n_tracks = -1
314    all_tracks = []
315    for idx, (frame, det_result) in enumerate(zip(my_video, detection_dict['detections'])):
316        ## Progress bar with estimated time remaining
317        if idx == 0:
318            start_loop = time.time()
319        elapsed_time = time.time() - start_loop
320        progress_bar(
321            idx,
322            len(my_video),
323            'Tracking Progress with currently %d tracks.%s' % (
324                n_tracks,
325                '(%ds left)' % (elapsed_time/idx*(len(my_video)-idx)) if idx else ''
326            )
327        )
328
329        ## Set image size once from the first frame
330        if idx == 0:
331            image_size = detection_dict['image_size'] if 'image_size' in detection_dict else frame.shape[:2][::-1]
332
333        ## Update tracker with the detection results
334        if tracker_type in ["bytetrack", "botsort"]:
335            # Bboxes in (x1, y1, x2, y2) format pixel coordinates and scores
336            bboxes = []
337            for det in det_result:
338                x, y, w, h = det['bbox']
339                x1 = int(x * image_size[0])
340                y1 = int(y * image_size[1])
341                x2 = int((x + w) * image_size[0])
342                y2 = int((y + h) * image_size[1])
343                bboxes.append([x1, y1, x2, y2])
344            bboxes = np.array(bboxes)
345            # Use "score" or "det_score" according to availability
346            scores = np.array([det['score'] for det in det_result])
347            if tracker_type == "bytetrack":
348                _current_tracks = tracker.update(bboxes, scores)
349            elif tracker_type == "botsort":
350                # Expecting bboxes in (x1, y1, x2, y2) format pixel coordinates, scores and classid (same classid for all detections here)
351                bboxes_scores = np.hstack((bboxes, scores[:, np.newaxis], np.ones((len(scores), 1)))) if len(scores) > 0 else np.empty((0, 6))
352                _current_tracks = tracker.update(bboxes_scores, frame)
353            # Convert back to (x, y, w, h) format and normalized coordinates
354            current_tracks = []
355            for track in _current_tracks:
356                x1, y1, w, h = track.tlwh
357                tmp = {}
358                tmp['bbox'] = [float(x1 / image_size[0]), float(y1 / image_size[1]), float(w / image_size[0]), float(h / image_size[1])]
359                tmp['track_id'] = track.track_id
360                tmp['score'] = float(track.score)
361                n_tracks = max(n_tracks, track.track_id)
362                current_tracks.append(tmp)
363            all_tracks.append(current_tracks)
364        elif tracker_type == "deepsort":
365            current_tracks, max_track_id = update_tracker(tracker, frame, feat_model, det_result)
366            n_tracks = max(n_tracks, max_track_id)
367            all_tracks.append(current_tracks)
368        elif tracker_type == "IoU":
369            ## Assign track_ids
370            track_id = process_detections(det_result, last_tracks, track_id, image_size, score_th, iou_threshold=0.3, log=log)
371            n_tracks = track_id - 1
372            ## Update Tracking
373            if len(tracking_buffer) > tracking_size:
374                tracking_buffer.pop(0)
375            tracking_buffer.append(det_result)
376            last_tracks = get_last_tracks(tracking_buffer)
377
378            ## Save and display results
379            all_tracks.append(det_result)
380    progress_bar(len(my_video), len(my_video), 'Tracking done in %ds with %d tracks' % (time.time() - start_time, n_tracks), log=log, completed=True)
381
382    # Saving
383    output_results = {'detections': all_tracks, 'format': 'xywh', 'image_size': image_size, 'n_tracks': n_tracks, 'tracker_type': tracker_type}
384    if 'detection_classes' in detection_dict:
385        output_results['detection_classes'] = detection_dict['detection_classes']
386    save_json_file(output_results, output_file)
387    return output_results

389def classify(detection_dict, my_video, output_file, class_database=None, class_threshold=0.5, image_size=None, device='cpu', log=None):
390    '''
391    Classify the tracks of the detected Baboons using a pre-trained classifier and a dictionary with extracted features from the tracks.
392
393    Args:
394        detection_dict: dict, the detection and tracking results
395        my_video: VideoFrameIterator, the video to process
396        output_file: str, the path to save the output file
397        class_database: str, the path to the classification dictionary
398        class_threshold: float, the threshold for class assignment
399        image_size: tuple, the size of the image (width, height)
400        device: str, the device to use for feature extraction (default 'cpu')
401        log: logger, the logger to print the information (default None)
402
403    Returns:
404        dict, the classification results
405    '''
406    # Initialization
407    start_time = time.time()
408    ## Check if the output file already exists
409    if os.path.exists(output_file):
410        print_and_log('Output file %s already exists. Loading existing file.' % (output_file), log=log)
411        return load_json_file(output_file)
412    ## Classification names
413    if class_database:
414        classes = sorted(os.listdir(class_database))
415        img_path_dict = build_image_paths_dict(class_database)  # Check if the classification dictionary is well formed
416    else:
417        classes = []
418        img_path_dict = {}
419    classes.append('NoID')
420    n_tracks = 0
421    model, transform = load_model_and_transform(device=device)
422    feature_database = build_feature_dict(model, transform, img_path_dict)
423
424    ## Check if detection_dict is filepath or dict
425    detection_dict = load_as_detection_dict(detection_dict, image_size=image_size, log=log)
426    class_dict = copy.deepcopy(detection_dict)
427
428    ## Check if Steps 1 and 2 are already done
429    score_file = output_file.replace('.json', '_with_scores.json')
430    if os.path.exists(score_file):
431        print_and_log('Score file %s already exists. Loading existing file with scores.' % (score_file), log=log)
432        track_class_dict = load_json_file(score_file)['track_class_dict']
433    else:
434        ## Step 1: Sort dict per track_id
435        track_dict = perso_format_to_trackid_format(class_dict['detections'])
436
437        ## Step 2: For each track, extract the features per image and get the class scores
438        track_class_dict = defaultdict(list)
439        for idx, (track_id, track_dets) in enumerate(track_dict.items()):
440            if idx == 0:
441                start_loop = time.time()
442            elapsed_time = time.time() - start_loop
443            progress_bar(
444                track_id,
445                len(track_dict),
446                'Classifying tracks.%s' % ('(%ds left)' % (elapsed_time/idx*(len(track_dict)-idx)) if idx > 0 else '')
447            )
448            features = []
449            idxs = []
450            for det in track_dets:
451                idx = det['image_id']-1  # image_id starts at 1 in coco format
452                idxs.append(idx)
453                img = my_video.get_frame_at_idx(idx)
454                x, y, w, h = det['bbox']
455                img_cropped = img[int(y*img.shape[0]):int((y+h)*img.shape[0]), int(x*img.shape[1]):int((x+w)*img.shape[1])]
456                feature = extract_feature(model, transform, img_cropped)
457                features.append(feature)
458            track_class_dict[track_id] = {
459                'scores': get_class_scores(features, feature_database),
460                'idxs': idxs
461            }
462        progress_bar(len(track_dict), len(track_dict), 'Classifying tracks done in %ds. Resolving assignments...' % (time.time() - start_time), log=log, completed=True)
463        # Save the intermediate results with the scores before resolving the final class assignments to avoid losing information in case of crash and for debugging purposes
464        save_json_file({'track_class_dict': track_class_dict, 'class_database': classes}, score_file)
465
466    ## Step 3: Final decision on the class of each track based on the scores, a threshold and overlapping tracks using while. If no score is above the threshold, assign "NoID" class.
467    final_assignments = resolve_class_assignments(track_class_dict, class_threshold=class_threshold)
468
469    ## Step 4: Save the results in in class_dict
470    class_to_idx = {cls_name: idx for idx, cls_name in enumerate(classes)}
471
472    for dets_per_frame in class_dict['detections']:
473        for det in dets_per_frame:
474            track_id = det['track_id']
475            assigned_class, assigned_score = final_assignments.get(track_id,("NoID", 0.0))
476            n_tracks = max(n_tracks, track_id)
477            # Save initial detection 
478            det['det_id'] = det.get('category_id', 1)
479            det['det_score'] = det.get('score')
480            # Reassign the class and score based on the classification results
481            det['category_id'] = class_to_idx[assigned_class]
482            det['score'] = assigned_score
483    # Saving
484    # class_dict['detection_classes'] = [f"Track {i}" for i in range(1, n_tracks + 1)]
485    class_dict['classification_classes'] = classes
486    save_json_file(class_dict, output_file)
487    print_and_log('Classification done in %ds for %d tracks. Results saved in %s' % (time.time() - start_time, n_tracks, output_file), log=log)
488
489    return class_dict

503def main(args, check_stop=false_check, gt_file_class_mot=None, log=None):

505    Main function to process the video.
506    Perform first the detection and tracking of the Baboons.
507    Then it classify each track using a Baboon dictionary with extracted features from the tracks and a pre-trained classifier.
508    Optionally, it creates a visualization of the tracks and the classification results on the video.
509
510    Args:
511        args: argparse.Namespace, the arguments
512        log: logger, the logger to print the information
513
514    Returns:
515        int, 1 if the function ran successfully
516    '''
517    # Path checks and output folder creation
518    if not os.path.exists(args.input_video):
519        print_and_log('Error: input %s must be a file or a folder' % (args.input_video), log=log, to_print=False)
520        raise ValueError('No input provided.')
521    os.makedirs(args.output, exist_ok=True)
522    
523    # Chrono

525
526    # Video object initialization
527    my_video = VideoFrameIterator(args.input_video, log=log)
528    image_size = my_video.get_image_size()
529    print_and_log('Video %s opened with resolution %s and %d frames.' % (args.input_video, str(image_size), len(my_video)), log=log)
530    # my_video.check_video()

532    # Load ground truth if evaluation is enabled

65def compute_iou(boxA, boxB, image_size):
66    '''
67    Compute the Intersection over Union (IoU) of two bounding boxes.
68
69    Args:
70        boxA: list of int, the first bounding box in the format [x, y, w, h]
71        boxB: list of int, the second bounding box in the format [x, y, w, h]
72        image_size: list of int, the size of the image in the format [width, height]
73
74    Returns:
75        float, the IoU value
76    '''
77    xA = max(boxA[0], boxB[0])*image_size[0]
78    yA = max(boxA[1], boxB[1])*image_size[1]
79    xB = min(boxA[0] + boxA[2], boxB[0] + boxB[2])*image_size[0]
80    yB = min(boxA[1] + boxA[3], boxB[1] + boxB[3])*image_size[1]
81    interArea = max(0, xB - xA) * max(0, yB - yA)
82    boxAArea = boxA[2] * boxA[3]* image_size[0] * image_size[1]
83    boxBArea = boxB[2] * boxB[3]* image_size[0] * image_size[1]
84    iou = interArea / float(boxAArea + boxBArea - interArea)
85    return iou

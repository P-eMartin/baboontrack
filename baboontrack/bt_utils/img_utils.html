 19class VideoFrameIterator:
 20    '''
 21    An iterator to iterate over the frames of a video file or image files.
 22    It supports both video files and image files such as .jpg, .png, .bmp, .tif, .zip files containing images.
 23    It can return a specific frame or iterate over all the frames.
 24    The iterator is reset automatically when the end of the video is reached.
 25    It contains information about the video such as fps, length, and whether it is a video or image files.
 26    It is advised to use the check_video method to check if the video is correctly loaded and to get the fps and length information.
 27    '''
 28    def __init__(self, path, video_extensions=('.mp4','.mkv','.avi','.mov', '.wmv'), img_extensions=('.jpeg','.jpg','.png','.bmp','.tif'), bgr=False, fps=None, log=None):
 29        '''
 30        Initialize the VideoFrameIterator.
 31
 32        Args:
 33            path: str, the path to the video file or image file or list of files
 34            video_extensions: tuple, the video file extensions to consider (default: .mp4, .mkv, .avi, .mov, .wmv)
 35            img_extensions: tuple, the image file extensions to consider (default: .jpeg, .jpg, .png, .bmp, .tif)
 36            bgr: bool, whether to convert the frames to BGR format (default False, which converts to RGB)
 37            fps: float, the frames per second used when processing image files (default None)
 38            log: logging.Logger, the logger to log the information (default None)
 39
 40        Returns:
 41            None
 42        '''
 43        self.path = path
 44        self.log = log
 45        self.bgr = bgr
 46        self.fps = fps
 47        # Check if the video path is a video file or directory/zip with image files
 48        if path.lower().endswith(video_extensions):
 49            self.cap = cv2.VideoCapture(path)
 50            # This may not work for all video formats
 51            self.length = int(self.cap.get(cv2.CAP_PROP_FRAME_COUNT))
 52            self.fps = self.cap.get(cv2.CAP_PROP_FPS)
 53            self.img = False
 54        else:
 55            # Create the list of image files to be processed
 56            if zipfile.is_zipfile(path):
 57                self.zip_file = zipfile.ZipFile(path, 'r')
 58                self.img_files = [f.filename for f in self.zip_file.infolist() if (f.filename.endswith(img_extensions)) and not os.path.basename(f.filename).split('/')[-1].startswith('.')]
 59            elif os.path.isdir(path):
 60                self.zip_file = None
 61                self.img_files = [os.path.join(path, f) for f in os.listdir(path) if f.endswith(img_extensions) and not f.startswith('.')]
 62            else:
 63                raise ValueError("The path %s is not a valid video file, or zip file or directory containing image files." % path)
 64            # Sort the files using natsort
 65            self.img_files = natsort.natsorted(self.img_files)
 66            self.length = len(self.img_files)
 67            self.img = True
 68            if self.fps is None:
 69                print_and_log('Cannot get fps from %s.' % (path), log=log)
 70        self.idx = 0
 71        self.count = 0
 72

 76    def __next__(self):
 77        if self.img:
 78            if self.idx >= self.length:
 79                # raise StopIteration
 80                self.reset_and_stop()
 81            frame = cv2.imread(self.img_files[self.idx])
 82        else:
 83            if not self.cap.isOpened():
 84                # raise StopIteration
 85                self.reset_and_stop()
 86
 87            ret, frame = self.cap.read()
 88            self.count += 1
 89            # Stop iteration when frame is not captured
 90            if not ret:
 91                if self.idx >= 0 or self.count >= self.length:
 92                    self.reset_and_stop()
 93                    # self.cap.release()
 94                    # raise StopIteration
 95                else:
 96                    # Call __next__ again to get the next frame
 97                    frame = self.__next__()
 98            # Convert the frame to RGB if needed (default)
 99            if not self.bgr:
100                frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
101        self.idx += 1
102        return frame

193    def check_video(self):
194        '''
195        Test the iterator by iterating over the video frames.
196        This method is useful to check if the video is correctly loaded.
197        '''
198        start_time = time.time()
199        for idx, frame in enumerate(self):
200            if idx % 100 == 0:
201                print_and_log("Frame %d/%d" % (idx, self.length), log=self.log)
202        total_frames = idx + 1
203        print_and_log("Video %s with fps %.2f iterated in %.2f seconds with a total frame of %d. Frames Processed PS: %.2f" % (
204            self.path,
205            self.fps,
206            time.time() - start_time,
207            total_frames,
208            total_frames / (time.time() - start_time)
209        ), log=self.log)
210        if self.length != total_frames:
211            print_and_log("Warning: The number of frames iterated (%d) is different from the video length (%d). Modifying video length information." % (total_frames, self.length), log=self.log)
212            self.length = total_frames
213            # check fps
214            fps_ffmpeg = self.get_fps_ffmpeg()
215            if fps_ffmpeg is not None and abs(fps_ffmpeg - self.fps) > 0.1:
216                print_and_log("Warning: The fps of the video (%f) is different from the one retrieved by ffmpeg (%f). Modifying fps information." % (self.fps, fps_ffmpeg), log=self.log)
217                self.fps = fps_ffmpeg
218            elif fps_ffmpeg is None:
219                print_and_log("Warning: Cannot retrieve fps from ffmpeg. Checking duration instead.", log=self.log)
220                duration_ffmpeg = self.get_duration_ffmpeg()
221                if duration_ffmpeg is not None:
222                    calculated_fps = self.length / duration_ffmpeg
223                    if abs(calculated_fps - self.fps) > 0.1:
224                        print_and_log("Warning: The fps of the video (%f) is different from the one calculated from the duration (%f). Modifying fps information." % (self.fps, calculated_fps), log=self.log)
225                        self.fps = calculated_fps
226                else:
227                    print_and_log("Warning: Cannot retrieve duration from ffmpeg. The fps information will not be modified.", log=self.log)
228            print_and_log("After check, considering video %s with fps %.2f with a total frame of %d." % (self.path, self.fps, self.length), log=self.log)

244    def get_duration_ffmpeg(self):
245        # Run the ffmpeg command to get video stream information
246        result = subprocess.run(
247            ['ffmpeg', '-i', self.path],
248            stderr=subprocess.PIPE,
249            stdout=subprocess.PIPE
250        )
251        
252        # Decode the output and search for the duration information
253        output = result.stderr.decode('utf-8')
254        match = re.search(r'Duration: (\d{2}:\d{2}:\d{2}\.\d{2})', output)
255        if match:
256            duration = match.group(1)
257            # Convert duration to seconds
258            h, m, s = map(float, duration.split(':'))
259            duration = h * 3600 + m * 60 + s
260            return duration
261        return None

263    def get_fps_ffmpeg(self):
264        # Run the ffmpeg command to get video stream information
265        result = subprocess.run(
266            ['ffmpeg', '-i', self.path],
267            stderr=subprocess.PIPE,
268            stdout=subprocess.PIPE
269        )
270        
271        # Decode the output and search for the FPS information
272        output = result.stderr.decode('utf-8')
273        for line in output.split('\n'):
274            if 'fps' in line:
275                # Extract the FPS value from the line
276                parts = line.split(',')
277                for part in parts:
278                    if 'fps' in part:
279                        fps = part.split('fps')[0].strip()

446def get_bbox(bbox, bbox_format='xywh', bbox_normalized=True, image_size=None):  
447    '''
448    Get the bounding box coordinates in xyxy format.
449
450    Args:
451        bbox: list, the bounding box coordinates
452        bbox_format: str, the format of the bounding box coordinates (default 'xywh')
453        bbox_normalized: bool, whether the bounding box coordinates are normalized (default True)
454        image_size: tuple, the size of the image (height, width) (default None, required if bbox_normalized is True)
455
456    Returns:
457        list: the bounding box coordinates in xyxy format
458    '''
459    if bbox_format == 'xywh':
460        x1 = bbox[0]
461        y1 = bbox[1]
462        x2 = bbox[0] + bbox[2]
463        y2 = bbox[1] + bbox[3]
464    elif bbox_format == 'xyxy':
465        x1, y1, x2, y2 = bbox
466    else:
467        raise ValueError("Invalid bbox_format %s. Should be 'xywh' or 'xyxy'." % (bbox_format))
468    if bbox_normalized:
469        if image_size is None:
470            raise ValueError("image_size should be provided when bbox_normalized is True.")
471        x1 = int(x1 * image_size[1])
472        y1 = int(y1 * image_size[0])
473        x2 = int(x2 * image_size[1])
474        y2 = int(y2 * image_size[0])
475    return [x1, y1, x2, y2]

478def write_text(frame, text, position, fontscale, color_text=None, color_bg=None, thickness=1, font=cv2.FONT_HERSHEY_SIMPLEX):
479    '''
480    Write text on a frame with a background.
481
482    Args:
483        frame: numpy array, the frame to write on
484        text: str, the text to write
485        position: tuple, the position to write the text (x, y)
486        fontscale: float, the scale of the font
487        color_text: tuple, the color of the text (default None, which is white if color_bg is None or dark, and black if color_bg is light)
488        color_bg: tuple, the color of the background (default None, which is transparent)
489        thickness: int, the thickness of the text (default 1)
490        font: int, the font to use (default cv2.FONT_HERSHEY_SIMPLEX)
491    Returns:
492        numpy array: the frame with the text written on it
493    '''
494    (text_width, text_height), baseline = cv2.getTextSize(text, font, fontscale, thickness)
495    x, y = position
496    
497    if color_bg is not None:
498        # Rectangle coordinates
499        ## Add a bit of padding for visual centering and adjust y to better center the text vertically
500        pad = int(0.1 * text_height)
501        rect_x1 = x - pad
502        rect_y1 = y - text_height - baseline - pad
503        rect_x2 = x + text_width + pad
504        rect_y2 = y + pad
505        text_height+= 2*pad
506        y-= int(baseline/2)
507        cv2.rectangle(frame, (rect_x1, rect_y1), (rect_x2, rect_y2), color_bg, -1)
508    if color_text is None:
509        if color_bg is not None:
510            # If the background color is light, use black text, otherwise use white text
511            if np.mean(color_bg) > 127:
512                color_text = (0, 0, 0)
513            else:
514                color_text = (255, 255, 255)
515        else:
516            color_text = (255, 255, 255)

698def enforce_min_bbox(bbox, frame_shape, min_size=50):
699    """
700    Ensure bbox is inside frame and has at least min_size.
701    
702    Args:
703        bbox: [x1, y1, x2, y2]
704        frame_shape: (H, W) or (H, W, C)
705        min_size: minimum width/height
706    
707    Returns:
708        [x1, y1, x2, y2] clipped and enlarged if needed
709    """
710    h, w = frame_shape[:2]
711    x1, y1, x2, y2 = map(int, bbox)
712    x1, x2 = sorted((x1, x2))
713    y1, y2 = sorted((y1, y2))
714    
715    # 1) If bbox already valid, return as-is
716    if x1 >= 0 and y1 >= 0 and x2 <= w-1 and y2 <= h-1 and (x2-x1) >= min_size and (y2-y1) >= min_size:
717        return [x1, y1, x2, y2]
718    
719    # 2) Clamp bbox inside frame
720    x1 = max(0, x1)
721    y1 = max(0, y1)
722    x2 = min(w-1, x2)
723    y2 = min(h-1, y2)
724    
725    # 3) Compute width and height after clamping
726    bbox_w = x2 - x1
727    bbox_h = y2 - y1
728    
729    # 4) Enlarge if width or height < min_size
730    if bbox_w < min_size or bbox_h < min_size:
731        cx = (x1 + x2) / 2
732        cy = (y1 + y2) / 2
733        half_w = max(bbox_w, min_size) / 2
734        half_h = max(bbox_h, min_size) / 2
735        
736        # Centered bbox
737        x1_new = int(round(cx - half_w))
738        x2_new = int(round(cx + half_w))
739        y1_new = int(round(cy - half_h))
740        y2_new = int(round(cy + half_h))
741        
742        # Shift bbox inside frame without unnecessary movement
743        x1_new = max(0, x1_new)
744        x2_new = min(w-1, x2_new)
745        y1_new = max(0, y1_new)
746        y2_new = min(h-1, y2_new)
747
748        # If after clamping the size < min_size, expand on the side that allows it
749        if (x2_new - x1_new) < min_size:
750            # Expand left if possible
751            x1_new = max(0, x2_new - min_size)
752            # Expand right if possible
753            x2_new = min(w-1, x1_new + min_size)
754
755        if (y2_new - y1_new) < min_size:
756            y1_new = max(0, y2_new - min_size)
757            y2_new = min(h-1, y1_new + min_size)
758        
759        x1, x2, y1, y2 = x1_new, x2_new, y1_new, y2_new
760    
761    return [x1, y1, x2, y2]

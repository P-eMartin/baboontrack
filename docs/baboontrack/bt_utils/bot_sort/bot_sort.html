436def joint_stracks(tlista, tlistb):
437    exists = {}
438    res = []
439    for t in tlista:
440        exists[t.track_id] = 1
441        res.append(t)
442    for t in tlistb:
443        tid = t.track_id
444        if not exists.get(tid, 0):
445            exists[tid] = 1
446            res.append(t)
447    return res

450def sub_stracks(tlista, tlistb):
451    stracks = {}
452    for t in tlista:
453        stracks[t.track_id] = t
454    for t in tlistb:
455        tid = t.track_id
456        if stracks.get(tid, 0):
457            del stracks[tid]
458    return list(stracks.values())

461def remove_duplicate_stracks(stracksa, stracksb):
462    pdist = iou_distance(stracksa, stracksb)
463    pairs = np.where(pdist < 0.15)
464    dupa, dupb = list(), list()
465    for p, q in zip(*pairs):
466        timep = stracksa[p].frame_id - stracksa[p].start_frame
467        timeq = stracksb[q].frame_id - stracksb[q].start_frame
468        if timep > timeq:
469            dupb.append(q)
470        else:
471            dupa.append(p)
472    resa = [t for i, t in enumerate(stracksa) if not i in dupa]
473    resb = [t for i, t in enumerate(stracksb) if not i in dupb]
474    return resa, resb

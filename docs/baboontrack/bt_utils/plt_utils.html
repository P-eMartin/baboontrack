 11def draw_annotation(ax, annotation, annotation_key, color, fontsize=12, linewidth=1, fy=1):
 12    '''
 13    Draw annotation on a Matplotlib Axes. It can be a point, a contour, a box or a bbox.
 14
 15    Args:
 16        ax: matplotlib.axes.Axes, the axes to draw the annotation on
 17        annotation: dict or list, annotation to draw
 18        annotation_key: str, key of the annotation
 19        color: color for the annotations
 20        fontsize: int, fontsize for the text
 21        linewidth: int, linewidth for the annotations
 22        fy: float, scaling factor for the annotation
 23
 24    Returns:
 25        elements: list, a list of Matplotlib elements added to the axes
 26    '''
 27    elements = []  # List to store the elements plotted
 28
 29    # Rescale annotation if needed
 30    if isinstance(annotation, dict):
 31        annotation = fy * np.array(list(annotation.values()))
 32    else:
 33        annotation = fy * np.array(annotation)
 34
 35    # Check the type of annotation and draw it
 36    if annotation_key == 'bbox':
 37        rect = plt.Rectangle(tuple(annotation[:2]), annotation[2] - annotation[0], annotation[3] - annotation[1],
 38                             edgecolor='green', facecolor='none', linewidth=linewidth)
 39        ax.add_patch(rect)
 40        elements.append(rect)
 41    elif annotation_key == 'box':
 42        line, = ax.plot(annotation[:, 0], annotation[:, 1], color='blue', linewidth=linewidth)
 43        elements.append(line)
 44    else:
 45        # Location
 46        text = annotation_key.replace('_', ' ')
 47        if len(annotation) > 0:  # If annotation is not empty
 48            if isinstance(annotation[0], np.ndarray):  # Region
 49                loc_mean = annotation.mean(axis=0, dtype=int)
 50                # Drawing contours
 51                if isinstance(annotation[0][0], np.ndarray):  # List of contours
 52                    for contour in annotation:
 53                        loc_mean = contour.mean(axis=0, dtype=int)
 54                        polygon = plt.Polygon(contour, closed=True, edgecolor=color, facecolor='none', linewidth=linewidth)
 55                        polygon_patch = ax.add_patch(polygon)
 56                        elements.append(polygon_patch)
 57                        # line, = ax.plot(contour[:, 0], contour[:, 1], color=color, linewidth=linewidth)
 58                        # elements.append(line)
 59                        txt = ax.text(loc_mean[0], loc_mean[1], text, fontsize=fontsize, color=color, ha='center')
 60                        elements.append(txt)
 61                else:  # Single contour
 62                    loc_mean = annotation.mean(axis=0, dtype=int)
 63                    polygon = plt.Polygon(annotation, closed=True, edgecolor=color, facecolor='none', linewidth=linewidth)
 64                    polygon_patch = ax.add_patch(polygon)
 65                    elements.append(polygon_patch)
 66                    # line, = ax.plot(annotation[:, 0], annotation[:, 1], color=color, linewidth=linewidth)
 67                    # elements.append(line)
 68                    txt = ax.text(loc_mean[0], loc_mean[1], text, fontsize=fontsize, color=color, ha='center')
 69                    elements.append(txt)
 70            else:  # Point
 71                point, = ax.plot(annotation[0], annotation[1], 'o', color=color, markersize=4)
 72                elements.append(point)
 73                txt = ax.text(annotation[0], annotation[1], text, fontsize=fontsize, color=color, ha='center')
 74                elements.append(txt)
 75
 76    return elements

 78def draw_annotation_on_img(img, annotation, annotation_key, colormap_annotation, fontface=cv2.FONT_HERSHEY_SIMPLEX, fontscale=.3, thickness=1, fy=1):
 79    '''
 80    Draw annotation on an image. It can be a point, a contour, a box or a bbox.
 81
 82    Args:
 83        img: np.array, image to draw the annotation on
 84        annotation: dict or list, annotation to draw
 85        annotation_key: str, key of the annotation
 86        colormap_annotation: dict, colormap for the annotations
 87    '''
 88    # Rescale annotation
 89    if isinstance(annotation, dict):
 90        annotation = (fy*np.array(list(annotation.values()))).astype(int)
 91    else:
 92        annotation = (fy*np.array(annotation)).astype(int)
 93    # Check the type of annotation and draw it
 94    if annotation_key == 'bbox':
 95        cv2.rectangle(img, tuple(annotation[:2]), tuple(annotation[2:]), (0, 255, 0), 2) #drawing bbox in Green
 96    elif annotation_key == 'box':
 97        cv2.drawContours(img,[annotation],0,(255,0,0), 2) #drawing bbox in Blue
 98    else:
 99        # Location
100        color = colormap_annotation[annotation_key]
101        text = annotation_key.replace('_',' ')
102        textsize = np.array(cv2.getTextSize(text, fontface, fontscale, thickness)[0])
103        if len(annotation)>0: # If annotation is not empty (can be empty for various reasons)
104            if isinstance(annotation[0],np.ndarray): # Region
105                loc_mean = annotation.mean(axis=0, dtype=int)
106                # Drawing contours
107                if isinstance(annotation[0][0],np.ndarray): # List of contours
108                    for contour in annotation:
109                        loc_mean = contour.mean(axis=0, dtype=int)
110                        cv2.drawContours(img, [contour], 0, color, thickness=2*thickness)
111                        cv2.putText(img, text=text, org=loc_mean-textsize//2, fontFace=fontface, fontScale=fontscale, color=color, thickness=thickness)
112                else: # Single contour
113                    loc_mean = annotation.mean(axis=0, dtype=int)
114                    cv2.drawContours(img, [annotation], 0, color, thickness=2*thickness) #drawing contours
115                    cv2.putText(img, text=text, org=loc_mean-textsize//2, fontFace=fontface, fontScale=fontscale, color=color, thickness=thickness)
116            else: # Point
117                cv2.circle(img, annotation, radius=4, color=color, thickness=thickness)
118                cv2.putText(img, text=text, org=annotation-textsize//2, fontFace=fontface, fontScale=fontscale, color=color, thickness=thickness)

120def plot_axes_in_new_fig(axes):
121    # Create a new figure and axis
122    fig, ax_new = plt.subplots()
123
124    # Extract images from the original axis
125    images = axes.get_images()
126    for image in images:
127        img_data = image.get_array()
128        ax_new.imshow(img_data, aspect=image.get_aspect(), extent=image.get_extent(), origin=image.get_origin())
129        ax_new.set_xlim(image.get_extent()[0], image.get_extent()[1])
130        ax_new.set_ylim(image.get_extent()[2], image.get_extent()[3])
131        ax_new.set_aspect(image.get_aspect())
132        ax_new.set_title(image.get_title())
133        ax_new.set_xlabel(image.get_xlabel())
134        ax_new.set_ylabel(image.get_ylabel())
135    # Copy the axis limits
136    ax_new.set_xlim(axes.get_xlim())
137    ax_new.set_ylim(axes.get_ylim())
138    # Copy the axis ticks
139    ax_new.set_xticks(axes.get_xticks())
140    ax_new.set_yticks(axes.get_yticks())
141    ax_new.set_xticklabels(axes.get_xticklabels())
142    ax_new.set_yticklabels(axes.get_yticklabels())
143
144    # Extract data from the original axis
145    lines = axes.get_lines()
146    for line in lines:
147        x_data = line.get_xdata()
148        y_data = line.get_ydata()
149        ax_new.plot(x_data, y_data, label=line.get_label())
150
151    # Copy labels and title
152    ax_new.set_xlabel(axes.get_xlabel())
153    ax_new.set_ylabel(axes.get_ylabel())
154    ax_new.set_title(axes.get_title())
155
156    # Show legend if it exists
157    if axes.get_legend():
158        ax_new.legend()
159
160    return fig

453def get_color_map(categories):
454    '''
455    Get a color map from a list of categories using the rainbow colormap.
456
457    Args:
458        categories: list, list of categories
459
460    Returns:
461        dict: dictionary of colors for each category
462    '''
463    # Dict of colors
464    dict_colors = {cat:color for cat, color in zip(categories, matplotlib.cm.rainbow(np.linspace(0, 1, len(categories))))}
465    return dict_colors

467def generate_colormap_with_legend(min_val=0, max_val=1, width=50, height=720, fontsize=12):
468    '''
469    Generate a JET colormap with a legend for the values.
470    
471    Args:
472        min_val: float, minimum value of the colormap
473        max_val: float, maximum value of the colormap
474        width: int, width of the colormap
475        height: int, height of the colormap
476        fontsize: int, fontsize of the legend
477    
478    Returns:
479        np.array: the colormap with the legend
480    '''

468def generate_colormap_with_legend(min_val=0, max_val=1, width=50, height=720, fontsize=12):
469    '''
470    Generate a JET colormap with a legend for the values.
471    
472    Args:
473        min_val: float, minimum value of the colormap
474        max_val: float, maximum value of the colormap
475        width: int, width of the colormap
476        height: int, height of the colormap
477        fontsize: int, fontsize of the legend
478    
479    Returns:
480        np.array: the colormap with the legend
481    '''
482    # Create an image with the jet colormap
483    colormap = np.linspace(1, 0, height).astype(np.float32)
484    colormap = np.repeat(colormap[:, np.newaxis], width, axis=1)
485    colormap_jet = cv2.applyColorMap((colormap * 255).astype(np.uint8), cv2.COLORMAP_JET)
486    
487    # Calculate the interval for the legend based on height and fontsize
488    num_steps = height // (fontsize * 3)  # Roughly 3 times the fontsize for each label
489    step_value = (max_val - min_val) / num_steps
490    # Find a step value with fewer digits
491    rounded_step_value = round(step_value, -int(np.floor(np.log10(step_value))))
492    
493    # Draw the legend on the colormap
494    font = cv2.FONT_HERSHEY_SIMPLEX
495    for i in range(num_steps + 1):
496        value = min_val + i * rounded_step_value
497        # Ensure the value does not exceed max_val
498        value = min(value, max_val)
499        text = f"{value:.2f}"
500        position = (5, int(height - (i * (height / num_steps))))
501        cv2.putText(colormap_jet, text, position, font, fontsize / 30, (255, 255, 255), 1, cv2.LINE_AA)
502    return colormap_jet

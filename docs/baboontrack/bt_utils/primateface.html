103    def analyze(
104        self,
105        image: InputType,
106        det_threshold: Optional[float] = None,
107        nms_threshold: Optional[float] = None,
108    ) -> List[Face]:
109        """Detect and analyze all primate faces in an image.
110
111        Args:
112            image: Input image as a file path (``str``/``Path``),
113                numpy array (BGR), or PIL Image.
114            det_threshold: Override default detection threshold.
115            nms_threshold: Override default NMS threshold.
116
117        Returns:
118            List of :class:`Face` objects, sorted by detection
119            confidence (highest first). Empty list if no faces detected.
120
121        Raises:
122            FileNotFoundError: If image path does not exist.
123            ValueError: If image cannot be loaded or decoded.
124        """
125        bgr = self._load_image(image)
126        h, w = bgr.shape[:2]
127        image_size = (w, h)
128
129        det_thr = det_threshold if det_threshold is not None else self.det_threshold
130        nms_thr = nms_threshold if nms_threshold is not None else self.nms_threshold
131
132        # Run detection
133        bboxes, scores = self._processor.detect_primates(
134            bgr, bbox_thr=det_thr, nms_thr=nms_thr
135        )
136        print(f"Detected {len(bboxes)} faces in the image.")
137
138        if len(bboxes) == 0:
139            return []
140
141        # Run pose estimation
142        if self._processor.pose_model is not None:
143            pose_results = self._processor.estimate_poses(bgr, bboxes)
144            keypoints_xy = pose_results.keypoints  # (N, 68, 2)
145            keypoint_scores = pose_results.keypoint_scores  # (N, 68)
146
147        # Build Face objects
148        faces: List[Face] = []
149        for i in range(len(bboxes)):
150            if self._processor.pose_model is not None:
151                kpts_with_vis = np.column_stack([
152                    keypoints_xy[i],
153                    keypoint_scores[i][:, np.newaxis],
154                ]).astype(np.float32)
155            else:
156                kpts_with_vis = np.zeros((0, 3), dtype=np.float32)
157
158            face = Face(
159                bbox=bboxes[i].astype(np.float32),
160                score=float(scores[i]),
161                keypoints=kpts_with_vis,
162                _image=bgr,
163                _image_size=image_size,
164                _embedding_fn=self._embedding_fn,
165                _embedding_backend=self.embedding_model,
166            )
167            faces.append(face)
168
169        faces.sort(key=lambda f: f.score, reverse=True)
170        return faces

236            if draw_keypoints:
237                for j in range(68):
238                    if scores[j] > 0.3:
239                        px, py = int(kpts[j, 0]), int(kpts[j, 1])
240                        cv2.circle(canvas, (px, py), 2, (0, 0, 255), -1)
241
242            if draw_skeleton:
243                # Jaw contour
244                for j in range(16):
245                    if scores[j] > 0.3 and scores[j + 1] > 0.3:
246                        p1 = (int(kpts[j, 0]), int(kpts[j, 1]))
247                        p2 = (int(kpts[j + 1, 0]), int(kpts[j + 1, 1]))
248                        cv2.line(canvas, p1, p2, (255, 200, 0), 1)
249                # Eye, brow, nose, mouth contours
250                for start, end in [
251                    (17, 22), (22, 27),  # eyebrows
252                    (27, 31),            # nose bridge
253                    (31, 36),            # nose base
254                    (36, 42), (42, 48),  # eyes
255                    (48, 60), (60, 68),  # mouth
256                ]:

259                            p1 = (int(kpts[j, 0]), int(kpts[j, 1]))

261                            cv2.line(canvas, p1, p2, (255, 200, 0), 1)

263            if show_pose:
264                _draw_pose_axes(canvas, face.keypoints, face._image_size, face.bbox)
265
266        if output is not None:
267            cv2.imwrite(str(output), canvas)
268
269        return canvas

272    def _load_image(image: InputType) -> np.ndarray:
273        """Convert various input types to BGR numpy array.
274
275        Args:
276            image: File path, numpy array, or PIL Image.
277
278        Returns:
279            BGR numpy array.
280
281        Raises:
282            FileNotFoundError: If path does not exist.
283            ValueError: If image cannot be decoded.
284            TypeError: If input type is not supported.
285        """
286        if isinstance(image, (str, Path)):
287            path = Path(image)
288            if not path.exists():
289                raise FileNotFoundError(f"Image not found: {path}")
290            bgr = cv2.imread(str(path))
291            if bgr is None:
292                raise ValueError(f"Could not decode image: {path}")
293            return bgr
294
295        if isinstance(image, np.ndarray):
296            if image.ndim == 2:
297                return cv2.cvtColor(image, cv2.COLOR_GRAY2BGR)
298            if image.ndim == 3 and image.shape[2] == 4:
299                return cv2.cvtColor(image, cv2.COLOR_RGBA2BGR)
300            if image.ndim == 3 and image.shape[2] == 3:
301                return image
302            raise ValueError(
303                f"Unexpected array shape: {image.shape}. "
304                "Expected (H, W), (H, W, 3), or (H, W, 4)."
305            )
306
307        # PIL Image
308        try:
309            import PIL.Image
310
311            if isinstance(image, PIL.Image.Image):
312                rgb = np.array(image.convert("RGB"))
313                return cv2.cvtColor(rgb, cv2.COLOR_RGB2BGR)
314        except ImportError:
315            pass
316
317        raise TypeError(
318            f"Unsupported image type: {type(image)}. "
319            "Expected str, Path, numpy array, or PIL Image."
320        )

123    def _get_aligned_crop(self) -> Optional[np.ndarray]:
124        """Align face to canonical 112x112 for ArcFace embedding.
125
126        Returns:
127            Aligned 112x112 BGR image, or None if alignment fails.
128        """
129        from ._processor import align_face
130        from ._constants import TARGET_LANDMARKS_5PT_112X112
131
132        landmarks_68 = self.keypoints[:, :2]
133        if landmarks_68.shape[0] != 68 or np.any(np.isnan(landmarks_68)):
134            return None
135        aligned, _ = align_face(
136            self._image,
137            landmarks_68,
138            output_size=112,
139            target_landmarks=TARGET_LANDMARKS_5PT_112X112,
140        )
141        return aligned
142
143    @functools.cached_property
144    def embedding(self) -> np.ndarray:
145        """Face embedding vector for re-identification.
146
147        For ArcFace, the face is aligned to a canonical 112x112 pose before
148        embedding extraction. Falls back to the raw crop if alignment fails.
149
150        Returns:
151            1-D numpy array (512-d for ArcFace, 1536-d for MegaDescriptor).
152
153        Raises:
154            RuntimeError: If no embedding model was configured.
155        """
156        if self._embedding_fn is None:
157            raise RuntimeError(
158                "No embedding model loaded. Initialize with: "
159                "PrimateFace(embedding_model='arcface')"
160            )
161        if self._embedding_backend == "arcface":
162            aligned = self._get_aligned_crop()
163            if aligned is not None:
164                return self._embedding_fn(aligned)
165        return self._embedding_fn(self.crop)

167    def verify(
168        self, other: "Face", threshold: float = 0.4
169    ) -> Tuple[bool, float]:
170        """Compare this face to another for identity verification.
171
172        Uses cosine distance between embeddings.
173
174        Args:
175            other: Another Face object to compare against.
176            threshold: Maximum cosine distance to consider same identity.
177                Default 0.4 works well for ArcFace.
178
179        Returns:
180            Tuple of ``(is_same_person, cosine_distance)``.
181        """
182        e1 = self.embedding
183        e2 = other.embedding
184        cos_sim = float(np.dot(e1, e2) / (np.linalg.norm(e1) * np.linalg.norm(e2) + 1e-8))
185        distance = 1.0 - cos_sim
186        return distance < threshold, distance

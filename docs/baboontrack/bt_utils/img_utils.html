 23class VideoFrameIterator:
 24    '''
 25    An iterator to iterate over the frames of a video file or image files.
 26    It supports both video files and image files such as .jpg, .png, .bmp, .tif, .zip files containing images.
 27    It can return a specific frame or iterate over all the frames.
 28    The iterator is reset automatically when the end of the video is reached.
 29    It contains information about the video such as fps, length, and whether it is a video or image files.
 30    It is advised to use the check_video method to check if the video is correctly loaded and to get the fps and length information.
 31    '''
 32    def __init__(self, path, video_extensions=('.mp4','.mkv','.avi','.mov', '.wmv'), img_extensions=('.jpeg','.jpg','.png','.bmp','.tif'), bgr=False, fps=None, log=None):
 33        '''
 34        Initialize the VideoFrameIterator.
 35
 36        Args:
 37            path: str, the path to the video file or image file or list of files
 38            video_extensions: tuple, the video file extensions to consider (default: .mp4, .mkv, .avi, .mov, .wmv)
 39            img_extensions: tuple, the image file extensions to consider (default: .jpeg, .jpg, .png, .bmp, .tif)
 40            bgr: bool, whether to convert the frames to BGR format (default False, which converts to RGB)
 41            fps: float, the frames per second used when processing image files (default None)
 42            log: logging.Logger, the logger to log the information (default None)
 43
 44        Returns:
 45            None
 46        '''
 47        self.path = path
 48        self.log = log
 49        self.bgr = bgr
 50        self.fps = fps
 51        # Check if the video path is a video file or directory/zip with image files
 52        if path.lower().endswith(video_extensions):
 53            self.cap = cv2.VideoCapture(path)
 54            # This may not work for all video formats
 55            self.length = int(self.cap.get(cv2.CAP_PROP_FRAME_COUNT))
 56            self.fps = self.cap.get(cv2.CAP_PROP_FPS)
 57            self.img = False
 58        else:
 59            # Create the list of image files to be processed
 60            if zipfile.is_zipfile(path):
 61                self.zip_file = zipfile.ZipFile(path, 'r')
 62                self.img_files = [f.filename for f in self.zip_file.infolist() if (f.filename.endswith(img_extensions)) and not os.path.basename(f.filename).split('/')[-1].startswith('.')]
 63            elif os.path.isdir(path):
 64                self.zip_file = None
 65                self.img_files = [os.path.join(path, f) for f in os.listdir(path) if f.endswith(img_extensions) and not f.startswith('.')]
 66            else:
 67                raise ValueError("The path %s is not a valid video file, or zip file or directory containing image files." % path)
 68            # Sort the files using natsort
 69            self.img_files = natsort.natsorted(self.img_files)
 70            self.length = len(self.img_files)
 71            self.img = True
 72            if self.fps is None:
 73                print_and_log('Cannot get fps from %s.' % (path), log=log)
 74        self.idx = 0
 75        self.count = 0

 81        if self.img:
 82            if self.idx >= self.length:
 83                # raise StopIteration

252            total_frames,
253            total_frames / (time.time() - start_time)
254        ), log=self.log)

264                print_and_log("Warning: Cannot retrieve fps from ffmpeg. Checking duration instead.", log=self.log)
265                duration_ffmpeg = self.get_duration_ffmpeg()
266                if duration_ffmpeg is not None:
267                    calculated_fps = self.length / duration_ffmpeg
268                    if abs(calculated_fps - self.fps) > 0.1:

271                else:
272                    print_and_log("Warning: Cannot retrieve duration from ffmpeg. The fps information will not be modified.", log=self.log)

289    def get_duration_ffmpeg(self):
290        # Run the ffmpeg command to get video stream information
291        result = subprocess.run(
292            ['ffmpeg', '-i', self.path],
293            stderr=subprocess.PIPE,
294            stdout=subprocess.PIPE
295        )
296        
297        # Decode the output and search for the duration information
298        output = result.stderr.decode('utf-8')
299        match = re.search(r'Duration: (\d{2}:\d{2}:\d{2}\.\d{2})', output)
300        if match:
301            duration = match.group(1)
302            # Convert duration to seconds
303            h, m, s = map(float, duration.split(':'))
304            duration = h * 3600 + m * 60 + s
305            return duration
306        return None

309        # Run the ffmpeg command to get video stream information
310        result = subprocess.run(

313            stdout=subprocess.PIPE
314        )
315        
316        # Decode the output and search for the FPS information
317        output = result.stderr.decode('utf-8')

502    frame, frame_annotations, image_size=None, detection_classes=None, classification_classes=None, track_color_dict=None,
503    detection_color_dict=None, classification_color_dict=None, thickness=2, fontscale=0.5, font=None, text_position="above",  # "above" or "below"

505):
506    """
507    Draw bounding boxes, labels, and masks on a frame.
508
509    Args:
510        frame: numpy array, the frame to draw on
511        frame_annotations: list of dicts, the annotations for the frame
512        image_size: tuple, the size of the image (height, width)
513        detection_classes: list, the detection classes (default None)
514        classification_classes: list, the classification classes (default None)
515        track_color_dict: dict, the color to use for plotting the tracks (default None)

523
524    Returns:

526    """
527
528    for ann in frame_annotations:
529        bbox = get_bbox(
530            ann["bbox"],
531            bbox_format=ann.get("bbox_format", "xywh"),
532            bbox_normalized=ann.get("bbox_normalized", True),
533            image_size=image_size,
534        )

536        track_id = ann.get("track_id")

616def apply_mask(frame, mask, color, alpha=0.5):
617    '''
618    Apply a mask to a frame with a given color and alpha.
619
620    Args:
621        frame: numpy array, the frame to apply the mask on
622        mask: numpy array, the binary mask to apply
623        color: tuple, the color to use for the mask (BGR format)
624        alpha: float, the alpha value to use for blending the mask (default 0.5)
625
626    Returns:
627        numpy array: the frame with the mask applied
628    '''
629     # Resize mask to match frame if needed
630    if mask.shape != frame.shape[:2]:
631        mask = cv2.resize(
632            mask,
633            (frame.shape[1], frame.shape[0]),
634            interpolation=cv2.INTER_NEAREST
635        )
636
637    # Create a color overlay for the mask
638    overlay = np.zeros_like(frame)
639    overlay[:] = color
640
641    # Blend the overlay with the original frame using the alpha value
642    blended = cv2.addWeighted(frame, 1 - alpha, overlay, alpha, 0)
643
644    # Apply the blended overlay only to the masked pixels
645    frame[mask == 1] = blended[mask == 1]
646
647    return frame

681def write_text(frame, text, position, fontscale, color_text=None, color_bg=None, thickness=1, font=cv2.FONT_HERSHEY_SIMPLEX):
682    '''
683    Write text on a frame with a background.
684
685    Args:
686        frame: numpy array, the frame to write on
687        text: str, the text to write
688        position: tuple, the position to write the text (x, y)
689        fontscale: float, the scale of the font
690        color_text: tuple, the color of the text (default None, which is white if color_bg is None or dark, and black if color_bg is light)
691        color_bg: tuple, the color of the background (default None, which is transparent)
692        thickness: int, the thickness of the text (default 1)
693        font: int, the font to use (default cv2.FONT_HERSHEY_SIMPLEX)
694    Returns:
695        numpy array: the frame with the text written on it
696    '''
697    (text_width, text_height), baseline = cv2.getTextSize(text, font, fontscale, thickness)
698    x, y = position
699    
700    if color_bg is not None:
701        # Rectangle coordinates
702        ## Add a bit of padding for visual centering and adjust y to better center the text vertically
703        pad = int(0.1 * text_height)
704        rect_x1 = x - pad
705        rect_y1 = y - text_height - baseline - pad
706        rect_x2 = x + text_width + pad
707        rect_y2 = y + pad
708        text_height+= 2*pad
709        y-= int(baseline/2)
710        cv2.rectangle(frame, (rect_x1, rect_y1), (rect_x2, rect_y2), color_bg, -1)
711    if color_text is None:
712        if color_bg is not None:
713            # If the background color is light, use black text, otherwise use white text
714            if np.mean(color_bg) > 127:
715                color_text = (0, 0, 0)
716            else:
717                color_text = (255, 255, 255)
718        else:
719            color_text = (255, 255, 255)
720    cv2.putText(frame, text, (x, y), font, fontscale, color_text, thickness)
721    return text_width, text_height

724def get_colormap_dict(classes, colormap=cv2.COLORMAP_VIRIDIS, cycle_size=None):
725    '''
726    Get a color dictionary for the given classes based on the given colormap.
727    Args:
728        classes: list, the list of classes to get the color dictionary for
729        colormap: int, the OpenCV colormap to use (default cv2.COLORMAP_VIRIDIS)
730        cycle_size: int, the size of the cycle for the colormap (default None, which is the number of classes)
731
732    Returns:
733        dict: the color dictionary for the given classes
734    '''
735    if classes is None or len(classes) == 0:
736        return {}
737    if cycle_size is None:
738        cycle_size = len(classes)
739    cycle_size = min(cycle_size, len(classes))
740    # Generate N colors
741    color_values = np.linspace(0, 255, cycle_size, dtype=np.uint8)[:, None]
742    colors = cv2.applyColorMap(color_values, colormap).reshape(-1, 3)
743    colors = [tuple(map(int, c)) for c in colors]
744    # Case when classes is a dict with key "name"
745    if isinstance(classes[0], dict) and "name" in classes[0]:
746        return {cls["name"]: colors[i % cycle_size] for i, cls in enumerate(classes)}
747    return {cls: colors[i % cycle_size] for i, cls in enumerate(classes)}

906def enforce_min_bbox(bbox, frame_shape, min_size=50):
907    """
908    Ensure bbox is inside frame and has at least min_size.
909    
910    Args:
911        bbox: [x1, y1, x2, y2]
912        frame_shape: (H, W) or (H, W, C)
913        min_size: minimum width/height
914    
915    Returns:
916        [x1, y1, x2, y2] clipped and enlarged if needed
917    """
918    h, w = frame_shape[:2]
919    x1, y1, x2, y2 = map(int, bbox)
920    x1, x2 = sorted((x1, x2))
921    y1, y2 = sorted((y1, y2))
922    
923    # 1) If bbox already valid, return as-is
924    if x1 >= 0 and y1 >= 0 and x2 <= w-1 and y2 <= h-1 and (x2-x1) >= min_size and (y2-y1) >= min_size:
925        return [x1, y1, x2, y2]
926    
927    # 2) Clamp bbox inside frame
928    x1 = max(0, x1)
929    y1 = max(0, y1)
930    x2 = min(w-1, x2)
931    y2 = min(h-1, y2)
932    
933    # 3) Compute width and height after clamping
934    bbox_w = x2 - x1
935    bbox_h = y2 - y1
936    
937    # 4) Enlarge if width or height < min_size
938    if bbox_w < min_size or bbox_h < min_size:
939        cx = (x1 + x2) / 2
940        cy = (y1 + y2) / 2
941        half_w = max(bbox_w, min_size) / 2
942        half_h = max(bbox_h, min_size) / 2
943        
944        # Centered bbox
945        x1_new = int(round(cx - half_w))
946        x2_new = int(round(cx + half_w))
947        y1_new = int(round(cy - half_h))
948        y2_new = int(round(cy + half_h))
949        
950        # Shift bbox inside frame without unnecessary movement
951        x1_new = max(0, x1_new)
952        x2_new = min(w-1, x2_new)
953        y1_new = max(0, y1_new)
954        y2_new = min(h-1, y2_new)
955
956        # If after clamping the size < min_size, expand on the side that allows it
957        if (x2_new - x1_new) < min_size:
958            # Expand left if possible
959            x1_new = max(0, x2_new - min_size)
960            # Expand right if possible
961            x2_new = min(w-1, x1_new + min_size)
962
963        if (y2_new - y1_new) < min_size:
964            y1_new = max(0, y2_new - min_size)
965            y2_new = min(h-1, y1_new + min_size)
966        
967        x1, x2, y1, y2 = x1_new, x2_new, y1_new, y2_new
968    
969    return [x1, y1, x2, y2]

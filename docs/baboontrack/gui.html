 116    class CreateToolTip(object):
 117        '''
 118        Create a tooltip for a given widget.
 119        '''
 120        def __init__(self, widget, text='widget info', waittime=300, wraplength=400):
 121            '''
 122            Initialize the CreateToolTip class.
 123
 124            Args:
 125                widget: tk.Widget, the widget to create a tooltip for
 126                text: str, the text of the tooltip
 127                waittime: int, waiting time before showing the tooltip in milliseconds (default: 300)
 128                wraplength: int, the width of the tipbox (default: 400)
 129            '''
 130            self.waittime = waittime
 131            self.wraplength = wraplength
 132            self.widget = widget
 133            self.text = text
 134            self.widget.bind("<Enter>", self.enter)
 135            self.widget.bind("<Leave>", self.leave)
 136            self.widget.bind("<ButtonPress>", self.leave)
 137            self.id = None
 138            self.tw = None
 139
 140        def enter(self, event=None):
 141            '''
 142            Show the tooltip.
 143
 144            Args:
 145                event: tk.Event, the event
 146            '''
 147            self.schedule()
 148
 149        def leave(self, event=None):
 150            '''
 151            Hide the tooltip.
 152
 153            Args:
 154                event: tk.Event, the event
 155            '''
 156            self.unschedule()
 157            self.hidetip()
 158
 159        def schedule(self):
 160            '''
 161            Schedule the tooltip.
 162            '''
 163            self.unschedule()
 164            self.id = self.widget.after(self.waittime, self.showtip)
 165
 166        def unschedule(self):
 167            '''
 168            Unschedule the tooltip.
 169            '''
 170            id = self.id
 171            self.id = None
 172            if id:
 173                self.widget.after_cancel(id)

 175        def showtip(self, event=None):
 176            '''
 177            Show the tooltip.
 178
 179            Args:
 180                event: tk.Event, the event
 181            '''
 182            x = y = 0
 183            x, y, cx, cy = self.widget.bbox("insert")
 184            # Lower corner of the widget
 185            x += self.widget.winfo_rootx() + self.widget.winfo_width()
 186            y += self.widget.winfo_rooty() + self.widget.winfo_height()
 187            # creates a toplevel window
 188            self.tw = tk.Toplevel(self.widget)
 189            # Leaves only the label and removes the app window
 190            self.tw.wm_overrideredirect(True)
 191            self.tw.wm_geometry("+%d+%d" % (x, y))
 192            label = tk.Label(self.tw, text=self.text, justify='left',
 193                        background="#ffffff", relief='solid', borderwidth=1,
 194                        wraplength = self.wraplength)
 195            label.pack(ipadx=1)

 197        def hidetip(self):
 198            '''
 199            Hide the tooltip.
 200            '''
 201            tw = self.tw
 202            self.tw= None
 203            if tw:
 204                tw.destroy()

 386    def create_image_fig(image_files, output_resolution):
 387        '''
 388        Create a Matplotlib figure with a mosaic of images.
 389
 390        Args:
 391            image_files: list, the list of image file paths
 392            output_resolution: tuple, the output resolution (width, height)
 393
 394        Returns:
 395            Figure, the Matplotlib figure
 396        '''
 397        # Load images
 398        images = [load_with_PIL(file) for file in image_files]
 399        
 400        # Calculate the number of rows and columns for the mosaic
 401        num_images = len(images)
 402        num_columns = math.ceil(math.sqrt(num_images))
 403        num_rows = math.ceil(num_images / num_columns)
 404        
 405        # Calculate the size of each cell in the mosaic
 406        mosaic_width, mosaic_height = output_resolution
 407        cell_width = mosaic_width // num_columns
 408        cell_height = mosaic_height // num_rows
 409        
 410        # Create a figure with subplots
 411        fig, axes = plt.subplots(num_rows, num_columns, figsize=(mosaic_width / 100, mosaic_height / 100))
 412        fig.subplots_adjust(top = 1, bottom = 0, right = 1, left = 0, hspace = 0, wspace = 0)
 413        
 414        # Flatten axes array for easy iteration
 415        axes = axes.flatten()
 416        
 417        # Position each image in the mosaic
 418        for index, img in enumerate(images):
 419            # Resize image to fit within its cell while maintaining aspect ratio
 420            img.thumbnail((cell_width, cell_height), Image.LANCZOS)
 421            
 422            # Display the image on the corresponding subplot
 423            axes[index].imshow(img)
 424            axes[index].axis('off')  # Hide the axes
 425            axes[index].margins(0,0)  # Remove margins
 426        
 427        # Hide any unused subplots
 428        for ax in axes[num_images:]:
 429            ax.axis('off')
 430        
 431        # Adjust layout
 432        plt.tight_layout()
 433        
 434        return fig

 437        '''
 438        Resize a figure while maintaining the aspect ratio.
 439
 440        Args:
 441            fig: Figure, the figure to resize
 442            max_width: int, the max width target of the figure
 443            max_height: int, the max height target of the figure
 444        '''
 445        # Get the original size of the figure
 446        original_width, original_height = fig.get_size_inches()
 447
 448        # Calculate the aspect ratio
 449        aspect_ratio = original_width / original_height
 450
 451        # Determine the new size according to the target size
 452        if max_width / max_height > aspect_ratio:
 453            new_width = max_height * aspect_ratio
 454            new_height = max_height
 455        else:
 456            new_width = max_width
 457            new_height = max_width / aspect_ratio
 458
 459        # Resize the figure
 460        fig.set_size_inches(new_width, new_height)

 463        '''
 464        Display an image or Matplotlib figure in the GUI.
 465

 468            img_ph: ttk.Frame, the placeholder for the image

 470            max_width: int, the maximum width of the image
 471        '''

 473        if max_height == 0 or max_height is None:
 474            max_height = 200
 475        if max_width == 0 or max_width is None:
 476            max_width = 200
 477
 478        # Differentiate between image and figure
 479        if isinstance(img_ori, Figure):
 480            fig = img_ori
 481            # Resize the figure to fit within the placeholder
 482            resize_figure(fig, max_width/100, max_height/100)
 483        elif isinstance(img_ori, str) and os.path.isdir(img_ori):
 484            imgs = [os.path.join(img_ori, f) for f in os.listdir(img_ori) if os.path.isfile(os.path.join(img_ori, f)) and f.lower().endswith(extensions)]
 485            if len(imgs) > max_mosaic:
 486                imgs = random.sample(imgs, max_mosaic-1) + [etc_path]
 487            if len(imgs) > 0:
 488                fig = create_image_fig(imgs, (max_width, max_height))
 489            else:
 490                warning_message = "Warning: No images found in the folder. Supported extensions are %s." % (', '.join(extensions))
 491                display_img(help_path, img_ph, max_height=max_height, max_width=max_width, warning_message=warning_message)
 492                return
 493        else:
 494            if isinstance(img_ori, str):
 495                if os.path.isfile(img_ori) and img_ori.endswith(extensions):
 496                    img = load_with_PIL(img_ori)
 497                elif img_ori == '':
 498                    img = load_with_PIL(none_path)
 499                else:
 500                    warning_message = "Warning: Not a valid Image. Supported extensions are %s." % (', '.join(extensions))
 501                    img = load_with_PIL(help_path)
 502            else:
 503                # Check number of channels to convert to RGBA
 504                if len(img_ori.shape) == 3 and img_ori.shape[2] == 3:
 505                    img = cv2.cvtColor(img_ori, cv2.COLOR_RGB2RGBA)
 506                else:
 507                    img = img_ori
 508                # img = Image.fromarray(img)
 509            
 510            # Create a Matplotlib figure
 511            if isinstance(img, Image.Image):
 512                img_width, img_height = img.size
 513            else:
 514                img_width, img_height = img.shape[1], img.shape[0]
 515
 516            resize_ratio = min(max_height/img_height, max_width/img_width)
 517
 518            if update_only:
 519                # Use the same size as the previous figure
 520                fig = img_ph.canvas.figure
 521                # Remove the previous elements
 522                if hasattr(fig, 'im_elements'):
 523                    for el in fig.im_elements:
 524                        try:
 525                            el.remove()
 526                        except:
 527                            pass
 528                if hasattr(fig, 'im'):
 529                    fig.im.remove()
 530                fig.set_size_inches(img_width*resize_ratio/100, img_height*resize_ratio/100)
 531            else:
 532                fig = Figure(figsize=(img_width*resize_ratio/100, img_height*resize_ratio/100))
 533                fig.ax = fig.add_subplot(111)
 534                fig.ax.axis('off')  # Hide axes
 535                fig.im_elements = []
 536                ## Set the limits of the image
 537                fig.ax.set_xlim(0, img_width)
 538                fig.ax.set_ylim(img_height, 0)
 539                fig.ax.margins(0,0)  # Remove margins
 540
 541            if colorbar is not None:
 542                # Display the image with a colorbar
 543                fig.im = fig.ax.imshow(img, cmap=colorbar['cmap'], vmin=colorbar['vmin'], vmax=colorbar['vmax'])
 544                # Add a colorbar
 545                fig.im_elements.append(fig.colorbar(fig.im, ax=fig.ax))
 546            else:
 547                # Display the image without a colorbar
 548                fig.im = fig.ax.imshow(img)
 549            # Draw annotations if provided
 550            if annotations is not None:
 551                for annotation_key in annotations.keys():
 552                    fig.im_elements.extend(
 553                            draw_annotation(
 554                                fig.ax,
 555                                annotations[annotation_key],
 556                                annotation_key,
 557                                color= annotations_color[annotation_key] if annotations_color is not None and annotation_key in annotations_color else 'white',
 558                                fontsize=8,
 559                                linewidth=1
 560                            )
 561                        )
 562            
 563        # Set the background of the figure to be transparent
 564        fig.patch.set_facecolor((0,0,0,0))
 565        # Tight layout
 566        fig.tight_layout()
 567        # No padding
 568        fig.subplots_adjust(left=0, right=1, top=1, bottom=0)
 569         
 570        # Update the canvas or create a new one
 571        if not hasattr(img_ph, 'canvas'):
 572            # Initialize the canvas with a figure of the same size than the target figure - to avoid wrong sizing with the first update
 573            img_ph.canvas = FigureCanvasTkAgg(Figure(figsize=(fig.get_size_inches()[0], fig.get_size_inches()[1]), facecolor=(0,0,0,0)), master=img_ph)
 574            img_ph.canvas.get_tk_widget().configure(background=bg)            
 575            img_ph.canvas.get_tk_widget().pack(side=tk.TOP, fill=tk.BOTH, expand=1, anchor='center')
 576            img_ph.canvas.get_tk_widget().update()
 577            
 578        # Close and Clear the previous figure
 579        if not update_only:
 580            plt.close(img_ph.canvas.figure)
 581            img_ph.canvas.figure.clear()
 582            img_ph.canvas.draw()
 583            img_ph.canvas.figure = fig
 584        # Update the size of the canvas
 585        img_ph.canvas.get_tk_widget().configure(width=fig.get_size_inches()[0]*100, height=fig.get_size_inches()[1]*100)
 586        img_ph.canvas.draw()
 587        img_ph.canvas.get_tk_widget().update()

 590        if warning_message is not None:
 591            messagebox.showwarning("Warning", warning_message, parent=root)

 123            Initialize the CreateToolTip class.
 124
 125            Args:

 129                wraplength: int, the width of the tipbox (default: 400)
 130            '''
 131            self.waittime = waittime
 132            self.wraplength = wraplength
 133            self.widget = widget
 134            self.text = text
 135            self.widget.bind("<Enter>", self.enter)
 136            self.widget.bind("<Leave>", self.leave)
 137            self.widget.bind("<ButtonPress>", self.leave)
 138            self.id = None
 139            self.tw = None

 141        def enter(self, event=None):
 142            '''
 143            Show the tooltip.
 144
 145            Args:
 146                event: tk.Event, the event
 147            '''
 148            self.schedule()
 149
 150        def leave(self, event=None):
 151            '''
 152            Hide the tooltip.
 153
 154            Args:
 155                event: tk.Event, the event
 156            '''
 157            self.unschedule()
 158            self.hidetip()
 159
 160        def schedule(self):
 161            '''
 162            Schedule the tooltip.
 163            '''
 164            self.unschedule()
 165            self.id = self.widget.after(self.waittime, self.showtip)
 166
 167        def unschedule(self):
 168            '''
 169            Unschedule the tooltip.
 170            '''
 171            id = self.id
 172            self.id = None
 173            if id:
 174                self.widget.after_cancel(id)

 177            '''
 178            Show the tooltip.
 179
 180            Args:
 181                event: tk.Event, the event
 182            '''
 183            x = y = 0

 185            # Lower corner of the widget
 186            x += self.widget.winfo_rootx() + self.widget.winfo_width()
 187            y += self.widget.winfo_rooty() + self.widget.winfo_height()
 188            # creates a toplevel window
 189            self.tw = tk.Toplevel(self.widget)
 190            # Leaves only the label and removes the app window
 191            self.tw.wm_overrideredirect(True)
 192            self.tw.wm_geometry("+%d+%d" % (x, y))
 193            label = tk.Label(self.tw, text=self.text, justify='left',

 387    def create_image_fig(image_files, output_resolution):
 388        '''
 389        Create a Matplotlib figure with a mosaic of images.
 390
 391        Args:
 392            image_files: list, the list of image file paths
 393            output_resolution: tuple, the output resolution (width, height)
 394
 395        Returns:
 396            Figure, the Matplotlib figure
 397        '''
 398        # Load images
 399        images = [load_with_PIL(file) for file in image_files]
 400        
 401        # Calculate the number of rows and columns for the mosaic
 402        num_images = len(images)
 403        num_columns = math.ceil(math.sqrt(num_images))
 404        num_rows = math.ceil(num_images / num_columns)
 405        
 406        # Calculate the size of each cell in the mosaic
 407        mosaic_width, mosaic_height = output_resolution
 408        cell_width = mosaic_width // num_columns
 409        cell_height = mosaic_height // num_rows
 410        
 411        # Create a figure with subplots
 412        fig, axes = plt.subplots(num_rows, num_columns, figsize=(mosaic_width / 100, mosaic_height / 100))
 413        fig.subplots_adjust(top = 1, bottom = 0, right = 1, left = 0, hspace = 0, wspace = 0)
 414        
 415        # Flatten axes array for easy iteration
 416        axes = axes.flatten()
 417        
 418        # Position each image in the mosaic
 419        for index, img in enumerate(images):
 420            # Resize image to fit within its cell while maintaining aspect ratio
 421            img.thumbnail((cell_width, cell_height), Image.LANCZOS)
 422            
 423            # Display the image on the corresponding subplot
 424            axes[index].imshow(img)
 425            axes[index].axis('off')  # Hide the axes
 426            axes[index].margins(0,0)  # Remove margins
 427        
 428        # Hide any unused subplots
 429        for ax in axes[num_images:]:
 430            ax.axis('off')
 431        
 432        # Adjust layout
 433        plt.tight_layout()
 434        
 435        return fig

 437    def resize_figure(fig, max_width, max_height):
 438        '''
 439        Resize a figure while maintaining the aspect ratio.
 440
 441        Args:
 442            fig: Figure, the figure to resize
 443            max_width: int, the max width target of the figure
 444            max_height: int, the max height target of the figure
 445        '''
 446        # Get the original size of the figure
 447        original_width, original_height = fig.get_size_inches()
 448
 449        # Calculate the aspect ratio
 450        aspect_ratio = original_width / original_height
 451
 452        # Determine the new size according to the target size
 453        if max_width / max_height > aspect_ratio:
 454            new_width = max_height * aspect_ratio
 455            new_height = max_height
 456        else:
 457            new_width = max_width
 458            new_height = max_width / aspect_ratio
 459
 460        # Resize the figure
 461        fig.set_size_inches(new_width, new_height)

 464        '''
 465        Display an image or Matplotlib figure in the GUI.
 466
 467        Args:
 468            img_ori: str, np.ndarray, or Figure, the image or figure to display
 469            img_ph: ttk.Frame, the placeholder for the image
 470            max_height: int, the maximum height of the image
 471            max_width: int, the maximum width of the image
 472        '''
 473        # Get the maximum size of the image
 474        if max_height == 0 or max_height is None:
 475            max_height = 200
 476        if max_width == 0 or max_width is None:
 477            max_width = 200
 478
 479        # Differentiate between image and figure
 480        if isinstance(img_ori, Figure):
 481            fig = img_ori
 482            # Resize the figure to fit within the placeholder
 483            resize_figure(fig, max_width/100, max_height/100)
 484        elif isinstance(img_ori, str) and os.path.isdir(img_ori):
 485            imgs = [os.path.join(img_ori, f) for f in os.listdir(img_ori) if os.path.isfile(os.path.join(img_ori, f)) and f.lower().endswith(extensions)]
 486            if len(imgs) > max_mosaic:
 487                imgs = random.sample(imgs, max_mosaic-1) + [etc_path]
 488            if len(imgs) > 0:
 489                fig = create_image_fig(imgs, (max_width, max_height))
 490            else:
 491                warning_message = "Warning: No images found in the folder. Supported extensions are %s." % (', '.join(extensions))
 492                display_img(help_path, img_ph, max_height=max_height, max_width=max_width, warning_message=warning_message)
 493                return
 494        else:
 495            if isinstance(img_ori, str):
 496                if os.path.isfile(img_ori) and img_ori.endswith(extensions):
 497                    img = load_with_PIL(img_ori)
 498                elif img_ori == '':
 499                    img = load_with_PIL(none_path)
 500                else:
 501                    warning_message = "Warning: Not a valid Image. Supported extensions are %s." % (', '.join(extensions))
 502                    img = load_with_PIL(help_path)
 503            else:
 504                # Check number of channels to convert to RGBA
 505                if len(img_ori.shape) == 3 and img_ori.shape[2] == 3:
 506                    img = cv2.cvtColor(img_ori, cv2.COLOR_RGB2RGBA)
 507                else:
 508                    img = img_ori
 509                # img = Image.fromarray(img)
 510            
 511            # Create a Matplotlib figure
 512            if isinstance(img, Image.Image):
 513                img_width, img_height = img.size
 514            else:
 515                img_width, img_height = img.shape[1], img.shape[0]
 516
 517            resize_ratio = min(max_height/img_height, max_width/img_width)
 518
 519            if update_only:
 520                # Use the same size as the previous figure
 521                fig = img_ph.canvas.figure
 522                # Remove the previous elements
 523                if hasattr(fig, 'im_elements'):
 524                    for el in fig.im_elements:
 525                        try:
 526                            el.remove()
 527                        except:
 528                            pass
 529                if hasattr(fig, 'im'):
 530                    fig.im.remove()
 531                fig.set_size_inches(img_width*resize_ratio/100, img_height*resize_ratio/100)
 532            else:
 533                fig = Figure(figsize=(img_width*resize_ratio/100, img_height*resize_ratio/100))
 534                fig.ax = fig.add_subplot(111)
 535                fig.ax.axis('off')  # Hide axes
 536                fig.im_elements = []
 537                ## Set the limits of the image
 538                fig.ax.set_xlim(0, img_width)
 539                fig.ax.set_ylim(img_height, 0)
 540                fig.ax.margins(0,0)  # Remove margins
 541
 542            if colorbar is not None:
 543                # Display the image with a colorbar
 544                fig.im = fig.ax.imshow(img, cmap=colorbar['cmap'], vmin=colorbar['vmin'], vmax=colorbar['vmax'])
 545                # Add a colorbar
 546                fig.im_elements.append(fig.colorbar(fig.im, ax=fig.ax))
 547            else:
 548                # Display the image without a colorbar
 549                fig.im = fig.ax.imshow(img)
 550            # Draw annotations if provided
 551            if annotations is not None:
 552                for annotation_key in annotations.keys():
 553                    fig.im_elements.extend(
 554                            draw_annotation(
 555                                fig.ax,
 556                                annotations[annotation_key],
 557                                annotation_key,
 558                                color= annotations_color[annotation_key] if annotations_color is not None and annotation_key in annotations_color else 'white',
 559                                fontsize=8,
 560                                linewidth=1
 561                            )
 562                        )
 563            
 564        # Set the background of the figure to be transparent
 565        fig.patch.set_facecolor((0,0,0,0))
 566        # Tight layout
 567        fig.tight_layout()
 568        # No padding
 569        fig.subplots_adjust(left=0, right=1, top=1, bottom=0)
 570         
 571        # Update the canvas or create a new one
 572        if not hasattr(img_ph, 'canvas'):
 573            # Initialize the canvas with a figure of the same size than the target figure - to avoid wrong sizing with the first update
 574            img_ph.canvas = FigureCanvasTkAgg(Figure(figsize=(fig.get_size_inches()[0], fig.get_size_inches()[1]), facecolor=(0,0,0,0)), master=img_ph)
 575            img_ph.canvas.get_tk_widget().configure(background=bg)            
 576            img_ph.canvas.get_tk_widget().pack(side=tk.TOP, fill=tk.BOTH, expand=1, anchor='center')
 577            img_ph.canvas.get_tk_widget().update()

 581            plt.close(img_ph.canvas.figure)
 582            img_ph.canvas.figure.clear()
 583            img_ph.canvas.draw()
 584            img_ph.canvas.figure = fig
 585        # Update the size of the canvas
 586        img_ph.canvas.get_tk_widget().configure(width=fig.get_size_inches()[0]*100, height=fig.get_size_inches()[1]*100)
 587        img_ph.canvas.draw()
 588        img_ph.canvas.get_tk_widget().update()
 589
 590        # Display the warning message if any
 591        if warning_message is not None:
 592            messagebox.showwarning("Warning", warning_message, parent=root)
